\newcommand\hoehe{194.8}% Ausschnitt: Oberkante der Tabelle der Handelszeiten
\newcommand\rand{24}%     Schatten: spread 4 + blur 20 px

% \zeile{<Oberkante>}{<dt-x>}{<dt-Breite>}{<Label>}{<Wert>}
\newcommand\zeile[5]{%
  \node[anchor=base west,inner sep=0,outer sep=0,text=label,font=\schrift\leicht]
    at (#2,#1+11.2) {#4};
  \node[anchor=base west,inner sep=0,outer sep=0,text=white]
    at (#2+#3,#1+11.2) {#5};}

% \marke{<Zeilenoberkante>}{<Nummer>}: Kreis 17 px, Mitte 3 px + Radius vor x150
\newcommand\marke[2]{%
  \fill[white] (138.5,#1+7) circle[radius=8.5];
  \node[anchor=base,inner sep=0,outer sep=0,text=markenziffer,
    font=\fontsize{6.954bp}{8bp}\halbfett] at (138.5,#1+7+3.091) {#2};}% 12 px * 17/22
