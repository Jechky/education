\newcommand{\gbz}[1]{\newline{\color{muted}\small #1}}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{30mm}%
  >{\raggedright\arraybackslash}p{102mm}>{\raggedright\arraybackslash}p{29.5mm}@{}}
\hd{Begriff} & \hd{Erklärung} & \hd{Mehr dazu} \\
\addlinespace[1.6mm]\midrule\addlinespace[2.4mm]
\textbf{Instrument} & Das handelbare Produkt, zum Beispiel ein Rohstoff oder eine Währung aus der Kategorie Forex (Devisen). In den
  Beispielen dieser Einheit steht dafür Instrument~XY. & \\ \addlinespace[2mm]
\textbf{Kurs} & Der aktuelle Preis eines Instruments. Er ändert sich laufend; diese Änderungen heißen
  Kursbewegungen. & \\ \addlinespace[2mm]
\textbf{Position}\gbz{(offen, geschlossen)} & Ein Handel in einem Instrument, den Sie öffnen und später
  schließen. Solange die Position offen ist, ändert sich ihr Ergebnis, also Gewinn oder Verlust, mit dem Kurs;
  mit dem Schließen wird es realisiert. & \verw{s:kontostand}{Kontostand} \\ \addlinespace[2mm]
\textbf{Buy / Sell} & Die Richtung einer Position: Buy (Kauf, auch long) oder Sell (Verkauf, auch short). Bei Buy
  entsteht mit steigendem Kurs ein Gewinn und mit fallendem ein Verlust, bei Sell umgekehrt. & \\ \addlinespace[2mm]
\textbf{Spread} & Der Unterschied zwischen dem Kurs für Buy und dem Kurs für Sell; er steht in den Angaben des
  Instruments. & \verw{s:instrumente}{Angaben eines Instruments} \\ \addlinespace[2mm]
\textbf{Volumen} & Der Wert einer Position. Gewinne und Verluste werden am gesamten Volumen berechnet.
  & \verw{s:hebel}{Hebel} \\ \addlinespace[2mm]
\textbf{Broker} & Das Unternehmen, das Ihr Konto führt und über das Sie Positionen öffnen und schließen. Anders
  als beim Direktkauf erwerben Sie das Instrument dabei nicht selbst. & \verw{s:hebel}{Hebel} \\ \addlinespace[2mm]
\textbf{Plattform} & Die Anwendung des Brokers, mobil und am Desktop. Dort öffnen und schließen Sie Positionen
  und sehen die Werte Ihres Kontos. & \verw{s:plattform}{In der Plattform} \\ \addlinespace[5mm]
\textbf{Margin} & Der Teil Ihres Kontos, den offene Positionen als Sicherheit binden, solange sie offen sind.
  & \verw{s:margin}{Margin} \\ \addlinespace[2mm]
\textbf{Stop out} & Die Schwelle des Levels, bei der die Plattform offene Positionen selbstständig schließt. Das
  Level gibt die Equity im Verhältnis zur Margin an. & \verw{s:margin}{Margin} \\ \addlinespace[2mm]
\textbf{Leverage}\gbz{(Hebel)} & Gibt an, welcher Teil des Volumens als Margin gebunden wird. Die Schreibweise
  1:10 bedeutet: Die Margin beträgt ein Zehntel des Volumens. & \verw{s:hebel}{Hebel}\newline
  \verw{s:instrumente}{Angaben eines Instruments} \\ \addlinespace[2mm]
\textbf{Swap} & Ein Betrag für das Halten einer Position über den Tageswechsel, der ihr je nach Satz (Swap long
  für Buy, Swap short für Sell) belastet oder gutgeschrieben wird. Er ist in ihrem Total Profit enthalten
  und geht beim Schließen in die Balance ein. & \verw{s:swap}{Swaps}\newline
  \verw{s:instrumente}{Angaben eines Instruments} \\ \addlinespace[2mm]
\textbf{Bonus} & Ein Betrag auf Ihrem Konto, mit dem Sie handeln können. Gewinne, die Sie damit erzielen,
  sind gemäß den AGB des Brokers auszahlbar. & \verw{s:plattform}{In der Plattform} \\
\end{tabular}

\kicker{}{Auf einen Blick}\phantomsection\label{s:ueberblick}
\Hone{Die Begriffe im Überblick}
\begin{minipage}{130mm}\lead{Die folgende Übersicht nennt für jeden Begriff seine Bedeutung und die Anlässe, zu denen sich sein Wert ändert.}\end{minipage}
\par\vspace{9mm}
% Spalten im Flattersatz; 28 + 72 + 61,5 mm plus Spaltenabstände = 170 mm (vorher 0,4 mm zu breit)
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{28mm}>{\raggedright\arraybackslash}p{72mm}%
  >{\raggedright\arraybackslash}p{61.5mm}@{}}
\hd{Begriff} & \hd{Bedeutung} & \hd{Änderung} \\
\addlinespace[1.6mm]\midrule\addlinespace[2.4mm]
\textbf{Balance} & Einzahlungen und Ergebnisse geschlossener Positionen & Bei Schließen sowie Ein- und Auszahlung \\ \addlinespace[2mm]
\textbf{Equity} & Balance zuzüglich Profit/Losses & Mit jeder Kursbewegung \\ \addlinespace[2mm]
\textbf{Profit/Losses} & Unrealisiertes Ergebnis der offenen Positionen & Mit jeder Kursbewegung \\ \addlinespace[2mm]
\textbf{Total Profit} & Ergebnis einer offenen Position einschließlich ihres Swaps & Mit jeder Kursbewegung
  und jedem Swap \\ \addlinespace[2mm]
\textbf{Margin} & Als Sicherheit gebundener Teil der Equity & Beim Öffnen und Schließen von Positionen \\ \addlinespace[2mm]
\textbf{Free} & Equity abzüglich Margin & Mit jeder Kursbewegung und jeder Position \\ \addlinespace[2mm]
\textbf{Level} & Equity im Verhältnis zur Margin & Mit jeder Kursbewegung und jeder Position \\ \addlinespace[2mm]
\textbf{Stop out} & Schwelle des Levels, an der die Plattform Positionen selbstständig schließt & Fest je Konto \\ \addlinespace[2mm]
\textbf{Leverage} & Verhältnis von Margin zu Volumen; bei 1:10 beträgt die Margin ein Zehntel des Volumens
  & Fest je Konto und Instrument \\ \addlinespace[2mm]
\textbf{Swap} & Belastung oder Gutschrift für das Halten über Nacht; bei der Position verbucht, beim Schließen
  in der Balance & Jede Nacht, am Mittwoch dreifach \\ \addlinespace[2mm]
\textbf{Bonus} & Betrag, mit dem Sie handeln können; Gewinne daraus sind gemäß den AGB auszahlbar
  & – \\
\end{tabular}
\par\vspace{6mm}
\begin{merk}\textbf{Kernaussagen.} Die Equity gibt den aktuellen Wert Ihres Kontos wieder, die Balance hingegen nur
die bereits realisierten Ergebnisse. Free bezeichnet den Teil der Equity, der für weitere Positionen zur Verfügung
steht. Der Hebel bestimmt nicht die Richtung eines Ergebnisses, sondern das Ausmaß, in dem eine Kursbewegung
auf Ihr Konto wirkt.\end{merk}
